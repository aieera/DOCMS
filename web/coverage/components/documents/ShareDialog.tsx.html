import { useState } from 'react'
import { Dialog } from '@/components/ui/Dialog'
import { Button } from '@/components/ui/Button'
import { Input } from '@/components/ui/Input'
import { Select } from '@/components/ui/Select'
import { Copy, Link2 } from 'lucide-react'
import toast from 'react-hot-toast'
 
interface Props { open: boolean; onOpenChange: (o: boolean) => void; documentId: string; documentTitle: string }
 
export function ShareDialog({ open, onOpenChange, documentTitle }: Props) {
  const [expiryDays, setExpiryDays] = useState('7')
  const [password, setPassword] = useState('')
  const [permission, setPermission] = useState('view')
  const [shareUrl, setShareUrl] = useState('')
 
  const createLink = () => {
    const token = crypto.randomUUID().slice(0, 12)
    setShareUrl(`${window.location.origin}/shared/${token}`)
    toast.success('Share link created')
  }
 
  const copyLink = () => {
    navigator.clipboard.writeText(shareUrl)
    toast.success('Link copied to clipboard')
  }
 
  return (
    <Dialog open={open} onOpenChange={onOpenChange} title={`Share "${documentTitle}"`} size="md">
      <div className="space-y-4">
        <Select label="Permission" value={permission} onValueChange={setPermission} options={[
          { value: 'view', label: 'View only' },
          { value: 'download', label: 'View + Download' },
          { value: 'edit', label: 'Edit' },
        ]} />
        <Input label="Password (optional)" type="password" value={password} onChange={(e) => setPassword(e.target.value)} placeholder="Leave empty for no password" />
        <Select label="Expires" value={expiryDays} onValueChange={setExpiryDays} options={[
          { value: '1', label: '1 day' },
          { value: '7', label: '7 days' },
          { value: '30', label: '30 days' },
          { value: '0', label: 'Never' },
        ]} />
        {shareUrl ? (
          <div className="flex items-center gap-2 rounded-md border border-[var(--color-border)] bg-slate-50 p-2 dark:bg-slate-800">
            <Link2 className="h-4 w-4 shrink-0 text-[var(--color-primary)]" />
            <code className="flex-1 truncate text-xs">{shareUrl}</code>
            <Button variant="ghost" size="sm" onClick={copyLink}><Copy className="h-4 w-4" /></Button>
          </div>
        ) : (
          <Button onClick={createLink} className="w-full">Create Share Link</Button>
        )}
      </div>
    </Dialog>
  )
}
