First tenant + admin:
  The chart creates no account (the old post-install seed registered a
  fixed admin@vaultdms.local / ChangeMe123!). Create the first tenant and
  admin once, against the owner DSN, with a strong password of your own:
    kubectl port-forward svc/{{ include "sedoc.dbHost" . }} 5432:5432 &
    cd scripts/seed && DATABASE_URL='postgres://<owner>:<pw>@localhost:5432/{{ .Values.global.database.name }}?sslmode=require' \
      SEED_ADMIN_EMAIL=<you@example.com> SEED_ADMIN_PASSWORD='<strong>' \
      SEED_TENANT_SLUG=<slug> SEED_TENANT_NAME='<name>' SEED_REGION={{ .Values.global.s3.region }} go run .

  kubectl scale deployment sedoc-document --replicas=3

  # View logs
  kubectl logs -f deployment/sedoc-document

  kubectl logs -f job/sedoc-migrate-{{ .Release.Revision }}

  # Trigger backup manually
  kubectl create job --from=cronjob/sedoc-backup sedoc-backup-manual
